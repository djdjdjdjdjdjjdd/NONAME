#include <iostream>


int main()
{
	int val1, val2;
	int result = 0;
	std::cout << "두개의 숫자 입력 : ";
	std::cin >> val1 >> val2;

	if (val1 < val2)
	{
		for (int i = val1 + 1; i < val2; i++)
			result += i;
	}
	else {
		for (int i = val2 + 1; i < val1; i++)
			result += i;
	}
	std::cout << "두 수 사이의 정수의합: " << result << std::endl;

	return 0;
}


<출력값>
두개의 숫자 입력:

함수의inline 선언

장점: 함수가 인라인화 되어 성능의 향상으로 이어질 수 있다

단점: 함수의 정의 방식이 일반함수에 비해서 복잡하다.
	따라서 복잡한 함수의 정의에는 한계가 있다.

inline 함수는 기존에 존재하는 함수 앞에 붙인다
예시)
int main()  - > inline int mai()

키워드 inline 선언은 컴파일러에 의해서 처리된다
따라서 컴파일러가 함수의 인라인화를 결정한다

inline 선언이 되어도 인라인 처리가 되지 않을수도 있다
inline 선언이 안되어있어도 인라인 처리가 될수도있다

inline함수는 매크로 함수와 달리 자료형에 독립적이지 못하다

참조자 의 이해
참조자 선언을 하고 있을떄 참조자는 꼭 초기화 시켜줘야한다

int num1=2010;
변수의 선언으로 인해서 num1 이라는 이름으로 메모리공간이 할당된다

int &num2 = num1
참조자의 선언으로 인해서 num1의 메모리공간에 num2라는 이름이 추가로 붙게된다

참조자는 기존에 선언된 변수에 붙이는 "별칭"이다 , 그리고 이렇게 참조자가
만들어지면 변수의 이름과 사실상 차이가 없다
======================================================
참조자 선언의 예제

int num1 = 2769; 
int &num2= num1;
int &num3= num2;

참조자의 수에는 제한이 없으며 , 참조자를 대상으로 참조자를 선언하는 것도 가능하다
======================================================
참조자의 선언가능 범위
예시)

int &ref = 20 ;
상수 대상으로의 참조선언은 불가능하다

int &ref;
참조자는 생성과 동시에 누군가를 참조행한다 (대상이 없어서 참조자 선언 x)

int &ref = NULL;
포인터처럼 NULL로 초기화 하는것도 불가능하다

변수의 성향을 지니는 대상이라면 참조자의 선언이 가능하다.
배열의 요소 역시 변수의 성향을 지니기 때문에 참조자의 선언이 가능하다
